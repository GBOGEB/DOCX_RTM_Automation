import os
import json
import yaml
import subprocess
from asciitree import LeftAligned
from asciitree.drawing import BoxStyle, BOX_LIGHT

# === Markdown Generation ===
def generate_markdown(data, output_path):
    """
    Generate Markdown documentation from data.
    """
    with open(output_path, "w", encoding="utf-8") as f:
        f.write("# Master Documentation\n\n")
        f.write("| Key | Value |\n")
        f.write("|-----|-------|\n")
        for key, value in data.items():
            f.write(f"| {key} | {value} |\n")
    print(f"✅ Markdown generated at {output_path}")

# === JSON/YAML Export ===
def export_json_yaml(data, json_path, yaml_path):
    """
    Export data to JSON and YAML formats.
    """
    with open(json_path, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=2)
    with open(yaml_path, "w", encoding="utf-8") as f:
        yaml.dump(data, f, sort_keys=False)
    print(f"✅ JSON and YAML exported at {json_path} and {yaml_path}")

# === ASCII Diagram Generation ===
def generate_ascii_diagram(data):
    """
    Generate an ASCII diagram from hierarchical data.
    """
    tr = LeftAligned(draw=BoxStyle(gfx=BOX_LIGHT, horiz_len=2))
    print(tr(data))

# === PANDOC Integration ===
def convert_markdown_to_docx(markdown_path, output_path):
    """
    Convert Markdown to Word (.docx) using PANDOC.
    """
    subprocess.run(["pandoc", markdown_path, "-o", output_path])
    print(f"✅ Converted {markdown_path} to {output_path}")

# === Main Workflow ===
def main():
    # Sample data
    data = {
        "Project": "Master Integration",
        "Author": "GitHub Copilot",
        "Description": "Integrated workflow for Python, Markdown, JSON, YAML, and more."
    }

    # Paths

    json_path = "../docs/master.json"
    yaml_path = "../docs/master.yaml"
    docx_path = "../docs/master.docx"

    # Generate outputs
    generate_markdown(data, markdown_path)
    export_json_yaml(data, json_path, yaml_path)
    convert_markdown_to_docx(markdown_path, docx_path)

import re

def lint_markdown(file_path):
    """
    Lint a Markdown file for common syntax issues.
    """
    with open(file_path, "r", encoding="utf-8") as f:
        lines = f.readlines()

    for i, line in enumerate(lines, start=1):
        if len(line) > 120:
            print(f"Line {i}: Exceeds 120 characters")
        if not re.match(r"^#+ ", line) and line.startswith("#"):
            print(f"Line {i}: Improper header formatting")

if __name__ == "__main__":
    lint_markdown("../docs/master.md")
